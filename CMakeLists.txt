cmake_minimum_required(VERSION 3.16)
project(ditserve VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(OpenSSL REQUIRED)

set(JSON_CACHE "${CMAKE_SOURCE_DIR}/.cache/deps/nlohmann3.11.3")
set(JSON_HEADER "${JSON_CACHE}/nlohmann/json.hpp")
set(JSON_SHA256 "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
if(NOT EXISTS "${JSON_HEADER}")
    file(MAKE_DIRECTORY "${JSON_CACHE}/nlohmann")
    file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp"
         "${JSON_HEADER}" EXPECTED_HASH "SHA256=${JSON_SHA256}" TLS_VERIFY ON STATUS JSON_DOWNLOAD_STATUS)
    list(GET JSON_DOWNLOAD_STATUS 0 JSON_DOWNLOAD_CODE)
    if(NOT JSON_DOWNLOAD_CODE EQUAL 0)
        file(REMOVE "${JSON_HEADER}")
        message(FATAL_ERROR "Cannot download pinned nlohmann JSON: ${JSON_DOWNLOAD_STATUS}")
    endif()
endif()
file(SHA256 "${JSON_HEADER}" JSON_ACTUAL_SHA256)
if(NOT JSON_ACTUAL_SHA256 STREQUAL JSON_SHA256)
    message(FATAL_ERROR "Cached nlohmann JSON 3.11.3 checksum mismatch")
endif()

set(CORE_MODULES types random cost workload backend policy policies metrics runtime)
foreach(KIND IN ITEMS src include)
    if(KIND STREQUAL "src")
        set(EXTENSION cpp)
    else()
        set(EXTENSION hpp)
    endif()
    file(GLOB_RECURSE MODULE_FILES CONFIGURE_DEPENDS RELATIVE "${CMAKE_SOURCE_DIR}/${KIND}/ditserve"
         "${KIND}/ditserve/*.${EXTENSION}")
    set(ACTUAL_MODULES)
    foreach(FILE IN LISTS MODULE_FILES)
        string(REGEX REPLACE "\\.${EXTENSION}$" "" MODULE "${FILE}")
        list(APPEND ACTUAL_MODULES "${MODULE}")
    endforeach()
    set(EXPECTED_MODULES ${CORE_MODULES})
    list(SORT ACTUAL_MODULES)
    list(SORT EXPECTED_MODULES)
    if(NOT ACTUAL_MODULES STREQUAL EXPECTED_MODULES)
        message(FATAL_ERROR "${KIND}/ditserve has missing/orphan modules; expected [${EXPECTED_MODULES}], found [${ACTUAL_MODULES}]. Every .cpp requires its matching .hpp.")
    endif()
endforeach()
set(CORE_SOURCES)
foreach(MODULE IN LISTS CORE_MODULES)
    list(APPEND CORE_SOURCES "src/ditserve/${MODULE}.cpp" "include/ditserve/${MODULE}.hpp")
endforeach()
add_library(ditserve_core ${CORE_SOURCES})
target_include_directories(ditserve_core PUBLIC include)
target_include_directories(ditserve_core SYSTEM PUBLIC "${JSON_CACHE}")
target_link_libraries(ditserve_core PUBLIC OpenSSL::Crypto)
target_compile_options(ditserve_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
add_executable(ditserve-core cmd/main.cpp)
target_link_libraries(ditserve-core PRIVATE ditserve_core)

include(CTest)
if(BUILD_TESTING)
    add_executable(core-tests tests/core/test_core.cpp)
    target_link_libraries(core-tests PRIVATE ditserve_core)
    target_compile_definitions(core-tests PRIVATE BASELINE_PATH="${CMAKE_SOURCE_DIR}/tests/fixtures/python-core-baseline.json")
    add_test(NAME core-contracts COMMAND core-tests)
endif()
